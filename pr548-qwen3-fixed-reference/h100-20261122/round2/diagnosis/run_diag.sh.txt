#!/bin/bash
source <HOME>/datastorage/q548/env.sh
D=$Q/handoff-197afeb5/h100-handoff/pr548-item4-20261119
REF=$SCRATCH/q548/results/reference/job-<JOBID>-attempt-0
CAND=$SCRATCH/q548/results/candidate/job-<JOBID>-attempt-0
OUT=$Q/out/diagnosis; ST=$Q/out/steps/steps.txt
cd $SRC; export PYTHONPATH=$SRC PYTHONNOUSERSITE=1
{ echo "src $(git rev-parse HEAD)"; echo "handoff $(git -C $Q/handoff-197afeb5 rev-parse HEAD)"; echo "status-before: [$(git status --porcelain --untracked-files=no)]"; $TP -c "import torch, sys; print(torch.__version__, sys.version.split()[0])"; nproc; } > $OUT/pre-checks.txt 2>&1
echo "[$(date -u +%FT%TZ)] START diag-1 (3 tensors x 5 runs)" >> $ST; s=$(date +%s)
$TP $D/diagnose_update_sensitivity.py --reference $REF/actor-fp32-0 \
  --other $REF/actor-fp32-1 --other $REF/actor-fp32-impl \
  --other $CAND/megatron-fp32-actor-tp2 --other $CAND/megatron-fp32-actor-tp4 \
  --copy-to $OUT/tensors --output $OUT/update-sensitivity.json > $OUT/update-sensitivity.log 2>&1
echo "[$(date -u +%FT%TZ)] END diag-1 rc=$? wall=$(( $(date +%s)-s ))s" >> $ST
echo "[$(date -u +%FT%TZ)] START diag-2 (scan TP4)" >> $ST; s=$(date +%s)
$TP $D/diagnose_update_sensitivity.py --reference $REF/actor-fp32-0 \
  --other $CAND/megatron-fp32-actor-tp4 --scan \
  --output $OUT/update-sensitivity-scan-tp4.json > $OUT/update-sensitivity-scan-tp4.log 2>&1
echo "[$(date -u +%FT%TZ)] END diag-2 rc=$? wall=$(( $(date +%s)-s ))s" >> $ST
git status --porcelain --untracked-files=no > $OUT/status-after.txt
touch $OUT/DONE
