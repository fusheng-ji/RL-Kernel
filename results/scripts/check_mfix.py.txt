def tail_counts(path):
    lines = [l for l in open(path, errors='replace').read().splitlines() if l.startswith('=') or ' in ' in l]
    last = lines[-1] if lines else ''
    c = {k: int(v) for v, k in re.findall(r'(\d+) (failed|passed|skipped|errors?|deselected)', last)}
    return c, last.strip('= ')
