# rec <dir> <checkout> <venv> <tag>: commit, torch/python line, tree status, other processes
rec() { local O=$1 C=$2 P=$3/bin/python T=$4; mkdir -p $O; git -C $C rev-parse HEAD > $O/commit-$T.txt
  $P -c "import torch, sys; print(torch.__version__, torch.version.cuda, sys.version)" > $O/env-$T.txt 2>&1
  git -C $C status --porcelain --untracked-files=no > $O/status-$T.txt
  ps -u $USER -o pid,etime,args | grep -E "python|torchrun|vllm|ray" | grep -vE "grep|ps -u" | cut -c1-150 > $O/other-processes-$T.txt; }
# cfile <dir> <checkout> <venv> <tag>: rl_engine._C.__file__ and rl_engine.__file__, and whether _C is inside the checkout
cfile() { (cd $2 && PYTHONPATH=$2 $3/bin/python -c "import rl_engine; print(rl_engine.__file__); import rl_engine._C as c; print(c.__file__)") > $1/c-file-$4.txt 2>&1
  case "$(realpath "$(tail -1 $1/c-file-$4.txt)" 2>/dev/null)" in $(realpath $2)/*) echo "inside" >> $1/c-file-$4.txt; return 0;; *) echo "NOT inside" >> $1/c-file-$4.txt; return 1;; esac; }
