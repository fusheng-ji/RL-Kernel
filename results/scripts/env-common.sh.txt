# rec <dir> <checkout> <python-or-venv> [tag]: commit, env line, tree status, other processes (nvidia-smi is taken on the node by smi)
rec() { local O=$1 W=$2 P=$3 T=${4:-before}; [ -d "$P" ] && P=$P/bin/python; mkdir -p $O; git -C $W rev-parse HEAD > $O/commit.txt
  $P -c "import torch, sys; print(torch.__version__, torch.version.cuda, sys.version)" > $O/env.txt 2>&1
  git -C $W status --porcelain --untracked-files=no > $O/status-$T.txt
  ps -u $USER -o pid,etime,args | grep -E "python|torchrun|vllm" | grep -vE "grep|ps -u" | cut -c1-150 > $O/other-processes-$T.txt; }
# smi <dir> <node> <tag>: nvidia-smi on the node, under the job
smi() { srun --jobid=$J -w $2 --overlap -N1 -n1 --mem=0 nvidia-smi > $1/nvidia-smi-$3.txt 2>&1; }
# cfile <dir> <checkout> <venv> <tag>: rl_engine._C.__file__, and whether it is inside the checkout
cfile() { (cd $2 && PYTHONPATH=$2 $3/bin/python -c "import rl_engine._C as c; print(c.__file__)") > $1/c-file-$4.txt 2>&1
  case "$(realpath "$(tail -1 $1/c-file-$4.txt)" 2>/dev/null)" in $(realpath $2)/*) echo "inside" >> $1/c-file-$4.txt; return 0;; *) echo "NOT inside" >> $1/c-file-$4.txt; return 1;; esac; }
