J = lambda p: json.load(open(p))


def walk(o, p=''):
    if isinstance(o, dict):
        for k, v in o.items():
            yield from walk(v, f'{p}.{k}' if p else k)
    elif isinstance(o, list):
        for i, v in enumerate(o):
            yield from walk(v, f'{p}[{i}]')
    else:
        yield p, o


def last(path):
    ls = [l for l in open(path, errors='replace').read().splitlines() if re.search(r'\d+ (passed|failed)', l)]
    return ls[-1].strip('= ') if ls else '(no summary line)'
