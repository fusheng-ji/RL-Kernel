#!/bin/bash
# Resume of qualify_reference.sbatch after an external kill (tool abort) at 2026-11-19 21:28 CET:
# identical commands, same result_dir, skipping the six actor runs that finished (actor-*-{0,1,impl}).
source <HOME>/datastorage/q548/env.sh
set -uo pipefail
cd $SRC; [[ -z $(git status --porcelain) ]] || { echo dirty; exit 3; }
result_dir=$(ls -d $SCRATCH/q548/results/reference/*/)
result_dir=${result_dir%/}
export PYTHONPATH=$SRC OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 OPENBLAS_NUM_THREADS=1 USE_TF=0 USE_FLAX=0 HF_HUB_OFFLINE=1 TOKENIZERS_PARALLELISM=false
O=$Q/out/steps; echo "[$(date -u +%FT%TZ)] RESUME reference (same commands; actor runs 0/1/impl fp32+bf16 already done)" >> $O/steps.txt
s0=$(date +%s)
for role in actor critic; do
  for precision in fp32 bf16; do
    for run in 0 1 impl; do
      [[ -f $result_dir/$role-$precision-$run/report.json && $role == actor ]] && continue
      control=(); [[ "$run" == impl ]] && control=(--implementation-control)
      s=$(date +%s)
      $TP -m tools.validation.models.qwen3_ppo.hf_ppo_reference --checkpoint "$CK" --rollout "$result_dir/rollout" --targets "$result_dir/targets" --role "$role" --precision "$precision" ${control[@]+"${control[@]}"} --output "$result_dir/$role-$precision-$run" > "$result_dir/$role-$precision-$run.log" 2>&1 || { echo "[$(date -u +%FT%TZ)] FAIL $role-$precision-$run" >> $O/steps.txt; exit 1; }
      echo "[$(date -u +%FT%TZ)] ok $role-$precision-$run $(( $(date +%s)-s ))s" >> $O/steps.txt
    done
    [[ -f $result_dir/$role-$precision-repeat-bounds.json ]] || $TP -m tools.validation.models.qwen3_ppo.compare_fixed --reference "$result_dir/$role-$precision-0" --other "$result_dir/$role-$precision-1" --extra-control "$result_dir/$role-$precision-impl" --output "$result_dir/$role-$precision-repeat-bounds.json" || { echo FAIL bounds >> $O/steps.txt; exit 1; }
  done
  $TP -m tools.validation.models.qwen3_ppo.compare_fixed --reference "$result_dir/$role-fp32-0" --other "$result_dir/$role-bf16-0" --extra-control "$result_dir/$role-bf16-impl" --output "$result_dir/$role-cross-precision-bounds.json" || { echo FAIL cross >> $O/steps.txt; exit 1; }
done
$TP -m tools.validation.models.qwen3_ppo.freeze_fixed_reference --reference-root "$result_dir" --output "$result_dir/acceptance.json" || { echo FAIL freeze >> $O/steps.txt; exit 1; }
echo "[$(date -u +%FT%TZ)] END reference-resume rc=0 wall=$(( $(date +%s)-s0 ))s" >> $O/steps.txt
touch $O/reference.DONE
